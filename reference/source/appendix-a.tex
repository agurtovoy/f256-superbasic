\compactappendixheader{Key bindings / shortcuts}

\endfirsthead
{\large\color{primary}\bfseries\sffamily Key / shortcut} & {\large\color{primary}\bfseries\sffamily Action} \\
\endhead
\\[0.6em]
\multicolumn{2}{l}{\textbf{Editing}} \\
\\[-0.2em]
\keybd{Enter} & Interpret the current line \\
\keybd{Shift}+\keybd{Enter} & Move to next line without interpreting \\
\keybd{Del} or \keybd{Backspace} & Delete character before cursor \\
\keybd{Ctrl}+\keybd{D} & Delete character at cursor \\
\keybd{Ctrl}+\keybd{K} & Delete characters from cursor position to line end \\
\keybd{Ins} & Insert a blank line \\
\\[0.9em]
\multicolumn{2}{l}{\textbf{Navigation}} \\
\\[-0.2em]
Arrow keys & Move cursor up / down / left / right \\
\keybd{Ctrl}+\keybd{Left} & Jump to previous word \\
\keybd{Ctrl}+\keybd{Right} & Jump to next word \\
\keybd{Ctrl}+\keybd{Up} or \keybd{Ctrl}+\keybd{A} & Jump to beginning of line \\
\keybd{Ctrl}+\keybd{Down} or \keybd{Ctrl}+\keybd{E} & Jump to end of line \\
\keybd{Ctrl}+\keybd{I} & Move cursor 8 characters to the right \\
\\[0.9em]
\multicolumn{2}{l}{\textbf{Scrolling}} \\
\\[-0.2em]
\keybd{WLD}+\keybd{Up} & Scroll screen up (show previous lines) \\
\keybd{WLD}+\keybd{Down} & Scroll screen down (show later lines) \\
\\[0.9em]
\multicolumn{2}{l}{\textbf{Other}} \\
\\[-0.2em]
\keybd{Clr} (\keybd{Shift}+\keybd{Home}) or \keybd{Ctrl}+\keybd{L} & Clear screen \\
\keybd{Run/Stop} or \keybd{Ctrl}+\keybd{C} & Break execution \\
\end{longtable}
